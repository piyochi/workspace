# サーバー・DB コンソールは自動で起動しない（先頭の @ は履歴に積むだけ。↑ で呼び出す）。起動・停止は開発ダッシュボードで行う

# Row 1: シェル
~/www/ticket_v2_third/ticket_v2_api | | env3 ticket_v2_api
~/www/ticket_v2_third/ticket_v2 | | env3 ticket_v2
~/www/ticket_v2_third | | env3 shell
~/www/ticket_v2_third/ticket_v2_api | @docker compose -f docker/docker-compose.yml --env-file docker/.env -p ticket_v2_api_3 up | env3 api server

# Row 2: DB
~/www/ticket_v2_third/ticket_v2_api | @docker compose -f docker/docker-compose.yml -p ticket_v2_api_3 run ticket_v2_api_app mysql ticket_v2_api_3_local -u root -h ticket_v2_api_db -proot | env3 primary
~/www/ticket_v2_third/ticket_v2_api | @docker compose -f docker/docker-compose.yml -p ticket_v2_api_3 run ticket_v2_api_app mysql ticket_v2_api_3_portal_local -u root -h ticket_v2_api_portal_db -proot | env3 portal
~/www/ticket_v2_third/ticket_v2_api | @docker compose -f docker/docker-compose.yml -p ticket_v2_api_3 run ticket_v2_api_app mysql ticket_v2_api_3_promoter_local_6 -u root -h ticket_v2_api_promoter_db -proot | env3 promoter
~/www/ticket_v2_third/ticket_v2 | @docker compose -f docker-compose.yml -p ticket_v2_api_3_f --env-file .env.development up | env3 app server

# Row 3: ワーカー・ログ
~/www/ticket_v2_third/ticket_v2_api | @docker exec -it ticket_v2_api_3_app bundle exec rake sqs:worker | env3 sqs
~/www/ticket_v2_third/ticket_v2_api | tail -f log/development.log | env3 api log
~/www/ticket_v2_third/ticket_v2 | tail -f logs/* | env3 app log
~/www/ticket_v2_third | | env3 free3
